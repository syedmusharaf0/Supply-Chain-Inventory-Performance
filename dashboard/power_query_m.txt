// ============================================================================
// Power Query (M) — load the three project tables into Power BI
// How to use: Power BI Desktop > Get Data > Blank Query > Advanced Editor,
// paste ONE block per table, name the query exactly as shown, Close & Apply.
// Set the DataFolder path in the first line of each block to your data folder.
// The CSV headers are already the friendly display names (Product Code,
// Warehouse Name, Units Sold, ...) so Power BI shows clean field names
// with no renaming step. DAX measures in dax_measures.dax use the same names.
// ============================================================================

// ---- Query name: dim_products ----
let
    DataFolder = "C:\path\to\supply-chain-inventory-performance\data\",
    Source = Csv.Document(File.Contents(DataFolder & "dim_products.csv"),
        [Delimiter = ",", Encoding = 65001, QuoteStyle = QuoteStyle.None]),
    Promoted = Table.PromoteHeaders(Source, [PromoteAllScalars = true]),
    Typed = Table.TransformColumnTypes(Promoted, {
        {"Product Code", type text}, {"Product Name", type text},
        {"Product Category", type text}, {"Supplier Name", type text},
        {"Unit Cost (INR)", type number}, {"Reorder Level", Int64.Type}})
in
    Typed

// ---- Query name: fact_daily_stock ----
let
    DataFolder = "C:\path\to\supply-chain-inventory-performance\data\",
    Source = Csv.Document(File.Contents(DataFolder & "fact_daily_stock.csv"),
        [Delimiter = ",", Encoding = 65001, QuoteStyle = QuoteStyle.None]),
    Promoted = Table.PromoteHeaders(Source, [PromoteAllScalars = true]),
    Typed = Table.TransformColumnTypes(Promoted, {
        {"Stock Date", type date}, {"Product Code", type text},
        {"Warehouse Name", type text}, {"Opening Stock (units)", Int64.Type},
        {"Units Received", Int64.Type}, {"Units Sold", Int64.Type},
        {"Lost Sales (units)", Int64.Type}, {"Closing Stock (units)", Int64.Type}})
in
    Typed

// ---- Query name: fact_purchase_orders ----
let
    DataFolder = "C:\path\to\supply-chain-inventory-performance\data\",
    Source = Csv.Document(File.Contents(DataFolder & "fact_purchase_orders.csv"),
        [Delimiter = ",", Encoding = 65001, QuoteStyle = QuoteStyle.None]),
    Promoted = Table.PromoteHeaders(Source, [PromoteAllScalars = true]),
    Typed = Table.TransformColumnTypes(Promoted, {
        {"Purchase Order ID", type text}, {"Order Date", type date},
        {"Product Code", type text}, {"Warehouse Name", type text},
        {"Quantity Ordered", Int64.Type}, {"Promised Lead Time (days)", Int64.Type},
        {"Actual Lead Time (days)", Int64.Type}, {"Delivery Date", type date}}),
    WithVariance = Table.AddColumn(Typed, "lead_time_variance_days", each
        [Actual Lead Time (days)] - [Promised Lead Time (days)], Int64.Type),
    WithLateFlag = Table.AddColumn(WithVariance, "is_late", each
        [lead_time_variance_days] > 0, type logical)
in
    WithLateFlag
